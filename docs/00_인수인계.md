```solidity
struct Quest {
    uint256 questId;
    QuestType questType;    // KILL, COLLECT, EXPLORE
    QuestDifficulty diff;   // F, E, D, C, B, A, S
    QuestCondition cond;    // 달성 조건
    QuestReward reward;     // SOUL + 아이템 + EXP
    address creator;        // 커뮤니티 제작자
    uint256 royaltyBps;     // 로열티 (500 = 5%)
}
// ORACLE_ROLE → updateQuestProgress()
// 완료 시 → EconomyEngine.mint() 호출
```
